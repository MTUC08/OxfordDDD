Bounded context: TicTacToe

workflow: StartGame
	Inputs: None
	Outputs: Grid (empty) and Player(starting)

workflow: MakeMove
Inputs: Player AND Position AND Grid 
Outputs: Grid(updated) AND Player(updated) AND GameState

data GameState = InProgress OR Tie OR (Win with Player)

data Player = X OR O
data Position = 
TopLeft OR TopMiddle OR TopRight OR MiddleLeft or Center 
OR MiddleRIght or BottomLeft or BottomMiddle or BottomRight

data Grid = list of (Position AND optional Player)
